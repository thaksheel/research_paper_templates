% Output style:  twocolumn | article | neurips | custom
\def\PaperStyle{twocolumn}

% --- NeurIPS only: options passed to neurips_2026.sty -----------------
%   main               main track, anonymous submission (line numbers)
%   main,final         camera-ready main track
%   preprint           arXiv version (non-anonymous, no line numbers)
%   position | eandd | creativeai | education
%   dblblindworkshop | sglblindworkshop   (also set the workshop title)
\def\NeurIPSOptions{main}
\def\NeurIPSWorkshopTitle{Name of the Workshop}

% --- Custom journal only: any .cls/.sty/.bst the journal provides ------
%   Put the journal's files in styles/ (or the main folder).
%   Examples:  elsarticle  [preprint,12pt]            bst: elsarticle-num
%              IEEEtran    [journal]                  bst: IEEEtran
%              informs4    [ijoc,blindrev]            bst: informs2014
%              svjour3     [smallextended]            bst: spmpsci
\def\CustomClass{article}
\def\CustomClassOptions{11pt}
\def\CustomStyle{}            % journal .sty without extension; empty = none
\def\CustomBibStyle{plainnat} % journal .bst without extension

% Bibliography database (ref.bib)
\def\BibFile{ref}

% --- Revision highlighting ---------------------------------------------
% Wrap changed text in \rev{...} or \begin{revision} ... \end{revision}.
%   \highlightrevisionstrue   -> changes shown in \RevisionColor
%   \highlightrevisionsfalse  -> changes typeset in black (final version)
\newif\ifhighlightrevisions
\highlightrevisionstrue
\def\RevisionColor{blue}      % any xcolor name, e.g. red, OliveGreen, blue!70!black

% Set to \LoadTheoremsfalse if the journal class already defines
% theorem/proof environments (e.g., svjour3, sn-jnl).
\newif\ifLoadTheorems
\LoadTheoremstrue

  \documentclass[10pt,twocolumn]{article}
  \usepackage{style-twocolumn}
  \def\PaperBibStyle{unsrtnat}
\else\ifx\PaperStyle\StyleArticle
  \documentclass[11pt]{article}
  \usepackage{style-article}
  \def\PaperBibStyle{plainnat}
\else\ifx\PaperStyle\StyleNeurIPS
  \documentclass{article}
  \usepackage{style-neurips}
  \def\PaperBibStyle{plainnat}
\else\ifx\PaperStyle\StyleCustom
  \edef\LoadPaperClass{\noexpand\documentclass[\CustomClassOptions]{\CustomClass}}
  \LoadPaperClass
  \ifx\CustomStyle\EmptyString\else
    \edef\LoadPaperStyle{\noexpand\usepackage{\CustomStyle}}
    \LoadPaperStyle
  \fi
  \let\PaperBibStyle\CustomBibStyle
\else
  \errmessage{Unknown \string\PaperStyle\space value. Use twocolumn, article, neurips or custom}
\fi\fi\fi\fi

% \begin{revision}...\end{revision}   longer block (equations, lists)

\newenvironment{revision}{\ifhighlightrevisions\color{\RevisionColor}\fi}{}
